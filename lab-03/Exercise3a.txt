#Modify ping sweep to allow me to type a subnet prefix and use that one in the script:

echo "Enter asubnet prefix: "
read subnet
ping_sweep() {
  for i in {1..10}; do
    ip="$subnet.$i"
    (ping -c1 -W1 $ip &>/dev/null && echo "[+] $ip alive") &
  done
  wait
}
