#run this in terminal to use, and execute with ping_sweep
ping_sweep() {
  subnet="172.17.0"
  for i in {1..10}; do
    ip="$subnet.$i"
    (ping -c1 -W1 $ip &>/dev/null && echo "[+] $ip alive") &
  done
  wait
}
